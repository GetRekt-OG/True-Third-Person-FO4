cmake_minimum_required(VERSION 4.3)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(TrueThirdPerson VERSION 0.2.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build an x64 Windows DLL with MSVC or clang-cl.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(COMMONLIB_RUNTIME_VR OFF CACHE BOOL "" FORCE)
set(COMMONLIB_RUNTIME_NG ON CACHE BOOL "" FORCE)
set(COMMONLIB_RUNTIME_OG ON CACHE BOOL "" FORCE)
set(COMMONLIB_RUNTIME_AE ON CACHE BOOL "" FORCE)
set(COMMONLIB_OPTION_FMT ON CACHE BOOL "" FORCE)
foreach(option XBYAK INI TOML JSON)
    set(COMMONLIB_OPTION_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_SHARED OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(fmt SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/fmt" OVERRIDE_FIND_PACKAGE)
FetchContent_Declare(spdlog SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/spdlog" OVERRIDE_FIND_PACKAGE)
FetchContent_MakeAvailable(fmt spdlog)
add_subdirectory(external/CommonLibF4 _deps/commonlib-build)
foreach(prop COMPILE_OPTIONS INTERFACE_COMPILE_OPTIONS)
    get_target_property(opts CommonLibF4 ${prop})
    list(REMOVE_ITEM opts "/arch:SSE4.2" "/Zc:templateScope" "/Zc:checkGwOdr" "/jumptablerdata" "/options:strict")
    set_property(TARGET CommonLibF4 PROPERTY ${prop} "${opts}")
endforeach()
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(CommonLibF4 PUBLIC -msse4.2)
endif()
add_library(TrueThirdPerson SHARED src/Main.cpp)
target_precompile_headers(TrueThirdPerson PRIVATE src/PCH.hpp)
target_link_libraries(TrueThirdPerson PRIVATE CommonLibF4::CommonLibF4)
set_target_properties(TrueThirdPerson PROPERTIES PREFIX "" OUTPUT_NAME "TrueThirdPerson")
install(TARGETS TrueThirdPerson RUNTIME DESTINATION Data/F4SE/Plugins COMPONENT TTP)
install(FILES package/Data/F4SE/Plugins/TrueThirdPerson.ini DESTINATION Data/F4SE/Plugins COMPONENT TTP)
install(DIRECTORY package/Data/Interface/ DESTINATION Data/Interface COMPONENT TTP)
install(DIRECTORY package/Data/MCM/ DESTINATION Data/MCM COMPONENT TTP)
